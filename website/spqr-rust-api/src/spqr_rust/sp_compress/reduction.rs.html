30fn needs_wide_child_refs_for_max(edge_count: usize, max_original_edge_id: u32) -> bool {

43fn widen_tree(tree: crate::sp_compress::types32::SpTree) -> SpTree {
44    let widen_child = |child: crate::sp_compress::types32::ChildRef| {
45        if crate::sp_compress::types32::child_is_macro(child) {
46            make_child_macro(u64::from(crate::sp_compress::types32::child_as_macro(
47                child,
48            )))
49        } else {
50            make_child_edge(crate::sp_compress::types32::child_as_edge(child))
51        }
52    };
53
54    SpTree {
55        macros: tree
56            .macros
57            .into_iter()
58            .map(|node| SpNode {
59                kind: node.kind,
60                _pad: node._pad,
61                left: node.left,
62                right: node.right,
63                children_offset: node.children_offset as u64,
64                children_count: node.children_count as u64,
65            })
66            .collect(),
67        children: tree.children.into_iter().map(widen_child).collect(),
68        core_edges: tree
69            .core_edges
70            .into_iter()
71            .map(|edge| CoreEdge {
72                u: edge.u,
73                v: edge.v,
74                child: widen_child(edge.child),
75            })
76            .collect(),
77        core_nodes: tree.core_nodes,
78        input_endpoints: tree.input_endpoints,
79        stats: tree.stats,
80    }
81}

113pub(crate) fn compress_borrowed_with_max_original_edge_id(
114    n_nodes: u32,
115    input_edges: &[InputEdge],
116    contractible: &[u8],
117    max_original_edge_id: u32,
118) -> CompressionResult {
119    if needs_wide_child_refs_for_max(input_edges.len(), max_original_edge_id) {

122        widen_result(crate::sp_compress::reduction32::compress_borrowed(
123            n_nodes,
124            input_edges,
125            contractible,

127    }

165fn compress_borrowed_impl(
166    n_nodes: u32,
167    input_edges: &[crate::sp_compress::types::InputEdge],
168    contractible: &[u8],
169    mut timings: Option<&mut CompressionTimings>,
170) -> CompressionResult {
171    macro_rules! add_timing {
172        ($field:ident, $start:expr) => {
173            if let Some(t) = timings.as_mut() {
174                t.$field += $start.elapsed().as_micros() as u64;
175            }
176        };
177    }
178
179    let mut tree = SpTree::default();
180    tree.stats.input_nodes = n_nodes;
181    tree.stats.input_edges = input_edges.len() as u32;
182
183    let t_input_edges = Instant::now();
184    tree.set_input_edges(input_edges);
185    add_timing!(t_input_edges_us, t_input_edges);
186
187    if contractible.len() < n_nodes as usize {
188        return CompressionResult {
189            tree,
190            success: false,
191            error_message: Some("contractible mask shorter than n_nodes"),
192        };
193    }
194
195    let n_nodes_usize = n_nodes as usize;
196    let t_init_work = Instant::now();
197
198    let mut arena = PNodeArena::new();
199    arena.reserve(input_edges.len() * 5 / 4 + 16);
200
201    let mut adj = AdjStore::new();
202
203    let mut pmap = FlatPairMap::new();
204    pmap.init(input_edges.len() + 16);
205
206    let pnode_start = arena.bulk_init_atomic(input_edges);
207
208    let mut edges: Vec<WorkEdge> = Vec::with_capacity(input_edges.len() * 5 / 4 + 16);
209    adj.init(n_nodes, input_edges.len());
210
211    for (k, ie) in input_edges.iter().enumerate() {
212        let pnode_id = pnode_start + k as u32;
213        let edge_idx = edges.len() as u32;
214        let adj_u = adj.insert(ie.u, edge_idx);
215        let adj_v = if ie.u != ie.v {
216            adj.insert(ie.v, edge_idx)
217        } else {
218            INVALID_ADJ
219        };
220        edges.push(WorkEdge {
221            u: ie.u,
222            v: ie.v,
223            pnode: pnode_id,
224            adj_node_u: adj_u,
225            adj_node_v: adj_v,
226            bucket_next: u32::MAX,
227        });
228
229        if ie.u != ie.v {
230            let k = make_pair_key(ie.u, ie.v);
231            let r = pmap.on_seen(k, edge_idx);
232            apply_on_seen(r, edge_idx, &mut edges);
233        }
234    }
235    add_timing!(t_init_work_us, t_init_work);
236
237    let t_init_dirty = Instant::now();
238    let mut node_dirty: Vec<NodeId> = Vec::new();
239    let mut node_in_dirty: Vec<u64> = vec![0; n_nodes_usize.div_ceil(64)];
240    for v_idx in 0..n_nodes_usize {
241        if contractible[v_idx] != 0 && adj.deg[v_idx] == 2 {
242            let bit = 1u64 << (v_idx & 63);
243            node_in_dirty[v_idx >> 6] |= bit;
244            node_dirty.push(NodeId(v_idx as u32));
245        }
246    }
247
248    let mut pair_dirty: Vec<PairKey> = Vec::new();
249    if !pmap.buckets.is_empty() {
250        for slot in pmap.slots.iter() {
251            let v = slot.value;
252            if FlatPairMap::is_indirect(v) {
253                let bid = FlatPairMap::bucket_index(v) as usize;
254                if pmap.buckets[bid].live_count() >= 2 {
255                    pair_dirty.push(slot.key);
256                }
257            }
258        }
259    }
260    add_timing!(t_init_dirty_us, t_init_dirty);
261
262    let mut series_reductions: u32 = 0;
263    let mut parallel_reductions: u32 = 0;
264
265    let mut bucket_edges_buf: Vec<u32> = Vec::with_capacity(64);
266    let mut kid_pnodes_buf: Vec<u32> = Vec::with_capacity(64);
267    let mut flat_pnodes_buf: Vec<u32> = Vec::with_capacity(64);
268
269    while !node_dirty.is_empty() || !pair_dirty.is_empty() {
270        let t_reduce_series = Instant::now();
271        while let Some(v) = node_dirty.pop() {
272            let v_idx = v.idx();
273            node_in_dirty[v_idx >> 6] &= !(1u64 << (v_idx & 63));
274
275            if contractible[v_idx] == 0 {
276                continue;
277            }
278            if adj.deg[v_idx] != 2 {
279                continue;
280            }
281
282            let (e1_idx, e2_idx) = adj.take_two(v);
283            if e1_idx == e2_idx {
284                continue;
285            }
286
287            let (e1_p, e1_u, e1_v, e1_au, e1_av) = {
288                let e = &edges[e1_idx as usize];
289                (e.pnode, e.u, e.v, e.adj_node_u, e.adj_node_v)
290            };
291            if e1_p == INVALID_PNODE || e1_u == e1_v {
292                continue;
293            }
294
295            let (e2_p, e2_u, e2_v, e2_au, e2_av) = {
296                let e = &edges[e2_idx as usize];
297                (e.pnode, e.u, e.v, e.adj_node_u, e.adj_node_v)
298            };
299            if e2_p == INVALID_PNODE || e2_u == e2_v {
300                continue;
301            }
302
303            let a = if e1_u == v { e1_v } else { e1_u };
304            let b = if e2_u == v { e2_v } else { e2_u };
305            if a == v || b == v {
306                continue;
307            }
308
309            let merged = arena.combine_series(v, a, b, e1_p, e2_p);
310
311            adj.remove(e1_u, e1_au);
312            if e1_u != e1_v {
313                adj.remove(e1_v, e1_av);
314            }
315            adj.remove(e2_u, e2_au);
316            if e2_u != e2_v {
317                adj.remove(e2_v, e2_av);
318            }
319            work_deactivate(&mut edges[e1_idx as usize]);
320            work_deactivate(&mut edges[e2_idx as usize]);
321
322            add_new_edge(
323                a,
324                b,
325                merged,
326                &mut edges,
327                &mut adj,
328                &mut pmap,
329                &mut node_dirty,
330                &mut node_in_dirty,
331                &mut pair_dirty,
332                contractible,
333                n_nodes_usize,
334            );
335            series_reductions += 1;
336        }
337        add_timing!(t_reduce_series_us, t_reduce_series);
338
339        let t_reduce_parallel = Instant::now();
340        while let Some(k) = pair_dirty.pop() {
341            let bid = match bucket_compact(&mut pmap, &mut edges, k) {
342                Some(b) => b,
343                None => continue,
344            };
345            if pmap.buckets[bid as usize].live_count() < 2 {
346                continue;
347            }
348
349            bucket_edges_buf.clear();
350            let mut cur = pmap.buckets[bid as usize].head;
351            while cur != u32::MAX {
352                let e = &edges[cur as usize];
353                let nxt = e.bucket_next;
354                if e.pnode != INVALID_PNODE {
355                    bucket_edges_buf.push(cur);
356                }
357                cur = nxt;
358            }
359            if bucket_edges_buf.len() < 2 {
360                continue;
361            }
362
363            let a = pair_first(k);
364            let c = pair_second(k);
365
366            kid_pnodes_buf.clear();
367            kid_pnodes_buf.reserve(bucket_edges_buf.len());
368            for &idx in &bucket_edges_buf {
369                kid_pnodes_buf.push(edges[idx as usize].pnode);
370            }
371
372            let merged =
373                arena.make_parallel_with_scratch(a, c, &kid_pnodes_buf, &mut flat_pnodes_buf);
374
375            for &idx in &bucket_edges_buf {
376                let (eu, ev, eau, eav) = {
377                    let e = &edges[idx as usize];
378                    (e.u, e.v, e.adj_node_u, e.adj_node_v)
379                };
380                adj.remove(eu, eau);
381                if eu != ev {
382                    adj.remove(ev, eav);
383                }
384                work_deactivate(&mut edges[idx as usize]);
385            }
386            pmap.erase_pair(k);
387
388            add_new_edge(
389                a,
390                c,
391                merged,
392                &mut edges,
393                &mut adj,
394                &mut pmap,
395                &mut node_dirty,
396                &mut node_in_dirty,
397                &mut pair_dirty,
398                contractible,
399                n_nodes_usize,
400            );
401            parallel_reductions += 1;
402        }
403        add_timing!(t_reduce_parallel_us, t_reduce_parallel);
404    }
405
406    let t_materialize = Instant::now();
407    let mut node_used: Vec<u64> = vec![0u64; n_nodes_usize.div_ceil(64)];
408
409    tree.children.reserve(input_edges.len());
410
411    let mut mat_stack: Vec<(u32, u8)> = Vec::with_capacity(64);
412    let mut mat_resolved: Vec<ChildRef> = Vec::with_capacity(64);
413    let mut mat_sort_keys: Vec<(u32, u32, ChildRef)> = Vec::with_capacity(64);
414
415    for i in 0..edges.len() {
416        let (epn, mut ce_u, mut ce_v) = {
417            let e = &edges[i];
418            if e.pnode == INVALID_PNODE {
419                continue;
420            }
421            (e.pnode, e.u, e.v)
422        };
423        node_used[ce_u.idx() >> 6] |= 1u64 << (ce_u.idx() & 63);
424        if ce_u != ce_v {
425            node_used[ce_v.idx() >> 6] |= 1u64 << (ce_v.idx() & 63);
426        }
427
428        let root_ref = materialize(
429            epn,
430            &mut arena,
431            &mut tree,
432            &mut mat_stack,
433            &mut mat_resolved,
434            &mut mat_sort_keys,
435        );
436
437        if ce_u.0 > ce_v.0 {
438            std::mem::swap(&mut ce_u, &mut ce_v);
439        }
440        tree.core_edges.push(CoreEdge {
441            u: ce_u.0,
442            v: ce_v.0,
443            child: root_ref,
444        });
445    }
446    add_timing!(t_materialize_us, t_materialize);
447
448    let t_cleanup = Instant::now();
449    arena.drop_storage();
450    let _ = std::mem::take(&mut edges);
451    adj.drop_storage();
452    pmap.drop_storage();
453    let _ = std::mem::take(&mut node_dirty);
454    let _ = std::mem::take(&mut node_in_dirty);
455    let _ = std::mem::take(&mut pair_dirty);
456    add_timing!(t_cleanup_us, t_cleanup);
457
458    let t_canon_series = Instant::now();
459    canonize_series_orientation(&mut tree);
460    add_timing!(t_canon_series_us, t_canon_series);
461
462    let t_sort_core_edges = Instant::now();
463    tree.core_edges.sort_unstable_by(|a, b| {
464        a.u.cmp(&b.u)
465            .then(a.v.cmp(&b.v))
466            .then(a.child.cmp(&b.child))
467    });
468    add_timing!(t_sort_core_edges_us, t_sort_core_edges);
469
470    let t_collect_core_nodes = Instant::now();
471    for v_idx in 0..n_nodes_usize {
472        if (node_used[v_idx >> 6] & (1u64 << (v_idx & 63))) != 0 {
473            tree.core_nodes.push(NodeId(v_idx as u32));
474        }
475    }
476    add_timing!(t_collect_core_nodes_us, t_collect_core_nodes);
477
478    let t_stats_shrink = Instant::now();
479    tree.stats.iterations = 1;
480    tree.stats.series_reductions = series_reductions;
481    tree.stats.parallel_reductions = parallel_reductions;
482    tree.stats.fully_sp_reducible =
483        if tree.core_edges.len() == 1 && tree.core_edges[0].u != tree.core_edges[0].v {
484            1
485        } else {
486            0
487        };
488
489    tree.update_stats();
490
491    add_timing!(t_stats_shrink_us, t_stats_shrink);
492
493    CompressionResult {
494        tree,
495        success: true,
496        error_message: None,
497    }
498}

500#[inline(always)]
501fn apply_on_seen(
502    result: crate::sp_compress::pmap::OnSeenResult,
503    edge_idx: u32,
504    edges: &mut [WorkEdge],
505) {
506    use crate::sp_compress::pmap::OnSeenResult;
507    match result {
508        OnSeenResult::SingleStored => {}
509        OnSeenResult::InsertedFirst { bucket_next, .. } => {
510            edges[edge_idx as usize].bucket_next = bucket_next;
511        }
512        OnSeenResult::PromotedAndInserted {
513            promoted_edge,
514            bucket_next,
515            ..
516        } => {
517            edges[promoted_edge as usize].bucket_next = u32::MAX;
518
519            edges[edge_idx as usize].bucket_next = bucket_next;
520        }
521    }
522}

524#[inline]
525fn enqueue_dirty_if_degree_two(
526    w: NodeId,
527    adj: &AdjStore,
528    node_dirty: &mut Vec<NodeId>,
529    node_in_dirty: &mut [u64],
530    contractible: &[u8],
531    n_nodes: usize,
532) {
533    let wi = w.idx();
534    if wi >= n_nodes {
535        return;
536    }
537    if contractible[wi] == 0 {
538        return;
539    }
540    if adj.deg[wi] != 2 {
541        return;
542    }
543    let bit = 1u64 << (wi & 63);
544    if (node_in_dirty[wi >> 6] & bit) != 0 {
545        return;
546    }
547    node_in_dirty[wi >> 6] |= bit;
548    node_dirty.push(w);
549}

551#[inline]
552#[allow(clippy::too_many_arguments)]
553fn add_new_edge(
554    u: NodeId,
555    v: NodeId,
556    pnode_id: u32,
557    edges: &mut Vec<WorkEdge>,
558    adj: &mut AdjStore,
559    pmap: &mut FlatPairMap,
560    node_dirty: &mut Vec<NodeId>,
561    node_in_dirty: &mut [u64],
562    pair_dirty: &mut Vec<PairKey>,
563    contractible: &[u8],
564    n_nodes: usize,
565) -> u32 {
566    let idx = edges.len() as u32;
567    let adj_u = adj.insert(u, idx);
568    let adj_v = if u != v {
569        adj.insert(v, idx)
570    } else {
571        INVALID_ADJ
572    };
573    edges.push(WorkEdge {
574        u,
575        v,
576        pnode: pnode_id,
577        adj_node_u: adj_u,
578        adj_node_v: adj_v,
579        bucket_next: u32::MAX,
580    });
581
582    if u != v {
583        let k = make_pair_key(u, v);
584        let r = pmap.on_seen(k, idx);
585        if matches!(
586            r,
587            crate::sp_compress::pmap::OnSeenResult::InsertedFirst { .. }
588                | crate::sp_compress::pmap::OnSeenResult::PromotedAndInserted { .. }
589        ) {
590            pair_dirty.push(k);
591        }
592        apply_on_seen(r, idx, edges);
593    }
594
595    enqueue_dirty_if_degree_two(u, adj, node_dirty, node_in_dirty, contractible, n_nodes);
596    if u != v {
597        enqueue_dirty_if_degree_two(v, adj, node_dirty, node_in_dirty, contractible, n_nodes);
598    }
599
600    idx
601}

603#[inline]
604fn bucket_compact(pmap: &mut FlatPairMap, edges: &mut [WorkEdge], k: PairKey) -> Option<u32> {
605    let bid = pmap.find_bucket(k)?;
606    pmap.mark_bucket_clean(bid);
607    let bid_us = bid as usize;
608    let mut cur = pmap.buckets[bid_us].head;
609    let mut new_head: u32 = u32::MAX;
610    let mut kept: u32 = 0;
611    while cur != u32::MAX {
612        let e = &mut edges[cur as usize];
613        let nxt = e.bucket_next;
614        if e.pnode != INVALID_PNODE {
615            e.bucket_next = new_head;
616            new_head = cur;
617            kept += 1;
618        }
619        cur = nxt;
620    }
621    pmap.buckets[bid_us].head = new_head;
622    pmap.buckets[bid_us].set_live_count(kept);
623    if kept == 0 {
624        pmap.erase_pair(k);
625        return None;
626    }
627    Some(bid)
628}

630fn materialize(
631    root_pnode: u32,
632    arena: &mut PNodeArena,
633    tree: &mut SpTree,
634    mat_stack: &mut Vec<(u32, u8)>,
635    mat_resolved: &mut Vec<ChildRef>,
636    mat_sort_keys: &mut Vec<(u32, u32, ChildRef)>,
637) -> ChildRef {
638    mat_stack.clear();
639    mat_stack.push((root_pnode, 0));
640
641    while let Some(&top) = mat_stack.last() {
642        let (p, phase) = top;
643
644        let (kind, left_kid, left, right) = {
645            let pn = &arena.pool[p as usize];
646            (pn.kind, pn.left_kid, pn.left, pn.right)
647        };
648
649        if kind == PK_ATOMIC {
650            mat_stack.pop();
651            continue;
652        }
653
654        if phase == 0 {
655            mat_stack.last_mut().unwrap().1 = 1;
656            let mut c = left_kid;
657            while c != INVALID_PNODE {
658                mat_stack.push((c, 0));
659                c = arena.pool[c as usize].next;
660            }
661            continue;
662        }
663
664        mat_resolved.clear();
665        let mut c = left_kid;
666        while c != INVALID_PNODE {
667            let cn = &arena.pool[c as usize];
668            let next = cn.next;
669            if cn.kind == PK_ATOMIC {
670                mat_resolved.push(make_child_edge(cn.edge_id));
671            } else {
672                let mm: SpNodeId = cn.edge_id.0 as SpNodeId;
673                mat_resolved.push(make_child_macro(mm));
674            }
675            c = next;
676        }
677
678        if kind == PK_PARALLEL && mat_resolved.len() > 1 {
679            let macros_snapshot: &[SpNode] = &tree.macros;
680            let children_snapshot: &[ChildRef] = &tree.children;
681            let first_edge_of = |r: ChildRef| -> u32 {
682                if child_is_edge(r) {
683                    return child_as_edge(r).0;
684                }
685                let mut cur = r;
686                while child_is_macro(cur) {
687                    let m = &macros_snapshot[child_as_macro(cur) as usize];
688                    if m.children_count == 0 {
689                        return EdgeId::INVALID.0;
690                    }
691                    cur = children_snapshot[m.children_offset as usize];
692                }
693                if child_is_edge(cur) {
694                    child_as_edge(cur).0
695                } else {
696                    EdgeId::INVALID.0
697                }
698            };
699
700            mat_sort_keys.clear();
701            mat_sort_keys.reserve(mat_resolved.len());
702            for &r in mat_resolved.iter() {
703                let kind_key = if child_is_edge(r) {
704                    0
705                } else {
706                    macros_snapshot[child_as_macro(r) as usize].kind as u32
707                };
708                mat_sort_keys.push((kind_key, first_edge_of(r), r));
709            }
710            mat_sort_keys.sort_unstable();
711            mat_resolved.clear();
712            mat_resolved.extend(mat_sort_keys.iter().map(|&(_, _, r)| r));
713        }
714
715        let children_offset = tree.children.len() as u64;
716        for &cr in mat_resolved.iter() {
717            tree.children.push(cr);
718        }
719
720        let m = SpNode {
721            kind: if kind == PK_SERIES {
722                SP_KIND_SERIES
723            } else {
724                SP_KIND_PARALLEL
725            },
726            _pad: [0; 3],
727            left: left.0,
728            right: right.0,
729            children_offset,
730            children_count: mat_resolved.len() as u64,
731        };
732
733        let new_mid = tree.macros.len() as SpNodeId;
734        tree.macros.push(m);
735
736        arena.pool[p as usize].edge_id = EdgeId(new_mid as u32);
737
738        mat_stack.pop();
739    }
740
741    let root = &arena.pool[root_pnode as usize];
742    if root.kind == PK_ATOMIC {
743        return make_child_edge(root.edge_id);
744    }
745    make_child_macro(root.edge_id.0 as SpNodeId)
746}
747
748fn canonize_series_orientation(tree: &mut SpTree) {
749    fn child_first_edge_id(tree: &SpTree, c: ChildRef) -> EdgeId {
750        if child_is_edge(c) {
751            return child_as_edge(c);
752        }
753        let mut cur_macro = child_as_macro(c);
754        loop {
755            let m = tree.macros[cur_macro as usize];
756            if m.children_count == 0 {
757                return EdgeId::INVALID;
758            }
759            let first = tree.children[m.children_offset as usize];
760            if child_is_edge(first) {
761                return child_as_edge(first);
762            }
763            cur_macro = child_as_macro(first);
764        }
765    }
766
767    for mid in 0..tree.macros.len() {
768        let m = tree.macros[mid];
769        if m.kind != SP_KIND_SERIES {
770            continue;
771        }
772
773        let reverse_it = match m.left.cmp(&m.right) {
774            std::cmp::Ordering::Greater => true,
775            std::cmp::Ordering::Equal => {
776                if m.children_count >= 2 {
777                    let first_child = tree.children[m.children_offset as usize];
778                    let last_child =
779                        tree.children[(m.children_offset + m.children_count - 1) as usize];
780                    let ef = child_first_edge_id(tree, first_child);
781                    let el = child_first_edge_id(tree, last_child);
782                    el < ef
783                } else {
784                    false
785                }
786            }
787            std::cmp::Ordering::Less => false,
788        };
789
790        if reverse_it {
791            let off = m.children_offset as usize;
792            let cnt = m.children_count as usize;
793            let mut a = 0;
794            let mut b = cnt - 1;
795            while a < b {
796                tree.children.swap(off + a, off + b);
797                a += 1;
798                b -= 1;
799            }
800            let mref = &mut tree.macros[mid];
801            std::mem::swap(&mut mref.left, &mut mref.right);
802        }
803    }
804}
