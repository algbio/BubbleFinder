 519         for (uint32_t tn = 0; tn < view_.numNodes; ++tn) {
 520             uint32_t s = view_.skeletonOffsets[tn];

 583     static std::unique_ptr<spqr_rust::RustSPQRResult> buildViaSpCompress_(
 584         const Graph& g, const uint8_t* contractible, uint32_t contractible_len)
 585     {
 586         const uint32_t n_nodes = static_cast<uint32_t>(g.numberOfNodes());
 587         const uint32_t n_edges = static_cast<uint32_t>(g.numberOfEdges());
 588         std::vector<SpCompressInputEdge> in_edges;
 589         in_edges.reserve(n_edges);
 590         for (uint32_t i = 0; i < n_edges; ++i) {
 591             uint32_t u = spqr_graph_edge_src(g.raw().raw(), i);
 592             uint32_t v = spqr_graph_edge_dst(g.raw().raw(), i);
 593             in_edges.push_back(SpCompressInputEdge{ u, v, i });
 594         }
 595         return std::make_unique<spqr_rust::RustSPQRResult>(
 596             n_nodes,
 597             in_edges.empty() ? nullptr : in_edges.data(),
 598             n_edges,
 599             n_edges == 0 ? 0u : n_edges - 1u,
 600             contractible,
 601             contractible_len);
 602     }

 615     class SkeletonGraph {
 616         const spqr_rust::SpqrTreeFlatView& view_;
 617         const Graph* gccGraph_;
 618         tree_node tn_;
 619         uint32_t nNodes_, edgeOff_, edgeEnd_;
 620         uint32_t mapOff_;
 621 
 622         mutable std::vector<uint32_t> adjStart_;
 623         mutable std::vector<uint32_t> adjEdges_;
 624         mutable std::vector<uint32_t> adjNeighbors_;
 625         mutable bool adjBuilt_ = false;
 626 
 627         std::pair<uint32_t, uint32_t> orientedEndpoints_(uint32_t skelIdx) const {
 628             const auto& se = view_.skeletonEdges[skelIdx];
 629             if (se.real_edge == UINT32_MAX || gccGraph_ == nullptr)
 630                 return {se.src, se.dst};
 631             auto gSrc = gccGraph_->source(::spqr::edge{se.real_edge});
 632             if (view_.nodeMapping[mapOff_ + se.src] == gSrc.idx)
 633                 return {se.src, se.dst};
 634             return {se.dst, se.src};
 635         }
 636 
 637         void buildAdj_() const {
 638             if (adjBuilt_) return;
 639             const uint32_t n = nNodes_;
 640             // 1) degree count
 641             adjStart_.assign(n + 1, 0);
 642             for (uint32_t i = edgeOff_; i < edgeEnd_; ++i) {
 643                 auto [s, d] = orientedEndpoints_(i);
 644                 ++adjStart_[s + 1];
 645                 if (d != s) ++adjStart_[d + 1];
 646             }
 647             // 2) prefix sum offsets
 648             for (uint32_t v = 0; v < n; ++v)
 649                 adjStart_[v + 1] += adjStart_[v];
 650             // 3) bucket fill
 651             const uint32_t total = adjStart_[n];
 652             adjEdges_.assign(total, 0);
 653             adjNeighbors_.assign(total, 0);
 654             std::vector<uint32_t> cursor(adjStart_.begin(), adjStart_.begin() + n);
 655             for (uint32_t i = edgeOff_; i < edgeEnd_; ++i) {
 656                 auto [s, d] = orientedEndpoints_(i);
 657                 uint32_t ps = cursor[s]++;
 658                 adjEdges_[ps] = i;
 659                 adjNeighbors_[ps] = d;
 660                 if (d != s) {
 661                     uint32_t pd = cursor[d]++;
 662                     adjEdges_[pd] = i;
 663                     adjNeighbors_[pd] = s;
 664                 }
 665             }
 666             adjBuilt_ = true;
 667         }

 747         void forEachEdge(F&& f) const {
 748             const auto& view = t_.view_;
 749             const uint32_t off = view.skeletonOffsets[tn_.idx];
 750             const uint32_t end = view.skeletonOffsets[tn_.idx + 1];
 751             const uint32_t mapOff = view.nodeMappingOffsets[tn_.idx];
 752             const Graph* gccGraph = t_.gccGraph_;
 753             for (uint32_t i = off; i < end; ++i) {
 754                 const auto& se = view.skeletonEdges[i];
 755                 uint32_t s = se.src, d = se.dst;
 756                 if (se.real_edge != UINT32_MAX && gccGraph != nullptr) {
 757                     auto gSrc = gccGraph->source(edge{se.real_edge});
 758                     if (view.nodeMapping[mapOff + s] != gSrc.idx) {
 759                         uint32_t tmp = s; s = d; d = tmp;
 760                     }
 761                 }
 762                 f(edge{i}, node{s}, node{d});
 763             }
 764         }

1241     // Precomputed O(1) virtual-edge index (see spqr::StaticSPQRTree for rationale).
1242     mutable std::unordered_map<uint64_t, uint32_t> virtualIndex_;
1243     mutable bool virtualIndexBuilt_ = false;
1244 
1245     static uint64_t packTreePair_(uint32_t from, uint32_t to) {
1246         return (static_cast<uint64_t>(from) << 32) | static_cast<uint64_t>(to);
1247     }
1248 
1249     void buildVirtualIndex_() const {
1250         if (virtualIndexBuilt_) return;
1251         virtualIndex_.reserve(view_.numNodes * 2);
1252         for (uint32_t tn = 0; tn < view_.numNodes; ++tn) {
1253             uint32_t s = view_.skeletonOffsets[tn];
1254             uint32_t e = view_.skeletonOffsets[tn + 1];
1255             for (uint32_t i = s; i < e; ++i) {
1256                 const auto& se = view_.skeletonEdges[i];
1257                 if (se.real_edge == UINT32_MAX) {
1258                     virtualIndex_.emplace(packTreePair_(tn, se.twin_tree_node), i);
1259                 }
1260             }
1261         }
1262         virtualIndexBuilt_ = true;
1263     }
1264 
1265     void buildTree() { parents_.resize(view_.numNodes); for (uint32_t i = 0; i < view_.numNodes; ++i) parents_[i] = view_.nodeParents[i]; tree_.build(view_.numNodes, parents_); }
1266     edge findVirtual(tree_node from, tree_node to) const {
1267         buildVirtualIndex_();
1268         auto it = virtualIndex_.find(packTreePair_(from.idx, to.idx));
1269         if (it == virtualIndex_.end()) return INVALID_EDGE;
1270         return edge{it->second - view_.skeletonOffsets[from.idx]};
1271     }
1272     edge findVirtualGlobal(tree_node from, tree_node to) const {
1273         buildVirtualIndex_();
1274         auto it = virtualIndex_.find(packTreePair_(from.idx, to.idx));
1275         if (it == virtualIndex_.end()) return INVALID_EDGE;
1276         return edge{it->second};

1347         mutable bool adjBuilt_ = false;
1348 
1349         void buildAdj_() const {
1350             if (adjBuilt_) return;
1351             const uint32_t n = nNodes_;
1352             adjStart_.assign(n + 1, 0);
1353             for (uint32_t i = edgeOff_; i < edgeEnd_; ++i) {
1354                 const auto& se = view_.skeletonEdges[i];
1355                 ++adjStart_[se.src + 1];
1356                 if (se.dst != se.src) ++adjStart_[se.dst + 1];
1357             }
1358             for (uint32_t v = 0; v < n; ++v)
1359                 adjStart_[v + 1] += adjStart_[v];
1360             const uint32_t total = adjStart_[n];
1361             adjEdges_.assign(total, 0);
1362             adjNeighbors_.assign(total, 0);
1363             std::vector<uint32_t> cursor(adjStart_.begin(), adjStart_.begin() + n);
1364             for (uint32_t i = edgeOff_; i < edgeEnd_; ++i) {
1365                 const auto& se = view_.skeletonEdges[i];
1366                 uint32_t ps = cursor[se.src]++;
1367                 adjEdges_[ps] = i;
1368                 adjNeighbors_[ps] = se.dst;
1369                 if (se.dst != se.src) {
1370                     uint32_t pd = cursor[se.dst]++;
1371                     adjEdges_[pd] = i;
1372                     adjNeighbors_[pd] = se.src;
1373                 }
1374             }
1375             adjBuilt_ = true;
1376         }

1492 // Check if directed graph is acyclic 
1493 inline bool isAcyclic(const Graph& G) {
1494     NodeArray<int> state(G, 0);  // 0=unvisited, 1=visiting, 2=done
1495     bool hasCycle = false;
1496     
1497     std::function<void(node)> dfs = [&](node u) {
1498         if (hasCycle) return;
1499         state[u] = 1;
1500         G.forEachAdj(u, [&](node v, edge e) {
1501             if (hasCycle) return;
1502             if (G.source(e) != u) return; 
1503             if (state[v] == 1) { hasCycle = true; return; }
1504             if (state[v] == 0) dfs(v);
1505         });
1506         state[u] = 2;
1507     };
1508     
1509     for (node v : G.nodes) {
1510         if (state[v] == 0) dfs(v);
1511         if (hasCycle) return false;
1512     }
1513     return true;
1514 }

1541 // Compute strongly CC (with Kosaraju algorithm)
1542 template<typename NA>
1543 inline int strongComponents(const Graph& G, NA& comp) {
1544     const uint32_t n = G.numberOfNodes();
1545     comp.init(G, -1);
1546     
1547     // First DFS to get finish order
1548     std::vector<node> order;
1549     order.reserve(n);
1550     NodeArray<bool> vis(G, false);
1551     
1552     std::function<void(node)> dfs1 = [&](node u) {
1553         vis[u] = true;
1554         G.forEachAdj(u, [&](node v, edge e) {
1555             if (G.source(e) != u) return;  // outgoing only
1556             if (!vis[v]) dfs1(v);
1557         });
1558         order.push_back(u);
1559     };
1560     
1561     for (node v : G.nodes) {
1562         if (!vis[v]) dfs1(v);
1563     }
1564     
1565     // Build reverse adjacency
1566     std::vector<std::vector<node>> radj(n);
1567     for (edge e : G.edges) {
1568         radj[G.target(e).idx].push_back(G.source(e));
1569     }
1570     
1571     // Second DFS on reverse graph in reverse finish order
1572     int numSCC = 0;
1573     std::function<void(node, int)> dfs2 = [&](node u, int c) {
1574         comp[u] = c;
1575         for (node v : radj[u.idx]) {
1576             if (comp[v] == -1) dfs2(v, c);
1577         }
1578     };
1579     
1580     for (int i = n - 1; i >= 0; --i) {
1581         node u = order[i];
1582         if (comp[u] == -1) {
1583             dfs2(u, numSCC++);
1584         }
1585     }
1586     
1587     return numSCC;
1588 }
