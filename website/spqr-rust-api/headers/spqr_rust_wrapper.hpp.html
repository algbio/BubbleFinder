 118     // Compute outdeg/indeg by counting (O(degree) per call)
 119     uint32_t outdeg(node v) const {
 120         uint32_t count = 0;
 121         uint32_t cursor = spqr_graph_adj_cursor(ptr_, v);
 122         node neighbor;
 123         edge e;
 124         uint32_t next;
 125         while (spqr_graph_adj_next(ptr_, cursor, &neighbor, &e, &next)) {
 126             if (spqr_graph_edge_src(ptr_, e) == v) ++count;
 127             cursor = next;
 128         }
 129         return count;
 130     }

 135         node neighbor;
 136         edge e;
 137         uint32_t next;
 138         while (spqr_graph_adj_next(ptr_, cursor, &neighbor, &e, &next)) {
 139             if (spqr_graph_edge_dst(ptr_, e) == v) ++count;
